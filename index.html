  <script>

    // --- STATE MANAGEMENT ---

    let isSimulatedExcess = false;


    const baseState = {

      sessionsCompleted: 42,

      fourWeekCadence: 2.75,

      runHours: 2.1,

      z12Minutes: 105,

      z45Minutes: 16,

      z3Minutes: 0,

      hrvLn: 3.92,

      deepSleepStr: "1h 22m",

      decoupling: 3.1

    };


    const excessState = {

      sessionsCompleted: 42,

      fourWeekCadence: 2.75,

      runHours: 3.4,     // Breaches 2.5 hr ceiling

      z12Minutes: 140,

      z45Minutes: 28,

      z3Minutes: 24,     // Breaches 15 min alert threshold

      hrvLn: 3.42,       // Below 3.65 lower corridor

      deepSleepStr: "42m",

      decoupling: 8.4    // Drift > 5.0%

    };


    function renderState(state) {

      // 1. Roadmap calculations

      document.getElementById('sessionCounter').innerText = state.sessionsCompleted;

      const b2Progress = ((state.sessionsCompleted - 25) / 40) * 100;

      document.getElementById('block2Fill').style.width = Math.min(100, Math.max(0, b2Progress)) + '%';

      

      // Estimated mass based on milestone kinetics

      const estGain = (0.9 + ((state.sessionsCompleted - 25) / 40) * 1.1).toFixed(2);

      document.getElementById('projectedGain').innerText = `+${estGain} kg Lean Mass`;


      // 2. Strength Cadence Floor

      document.getElementById('cadenceVal').innerText = state.fourWeekCadence.toFixed(2);

      document.getElementById('cadenceBar').style.width = Math.min(100, (state.fourWeekCadence / 3.0) * 100) + '%';


      // 3. Running Volume Ceiling (Cap: 2.5 hrs)

      document.getElementById('runTimeVal').innerText = state.runHours.toFixed(1);

      const runPercent = (state.runHours / 2.5) * 100;

      const runBar = document.getElementById('runTimeBar');

      const runAlert = document.getElementById('runTimeAlert');


      if (state.runHours > 2.5) {

        runBar.style.width = '100%';

        runBar.className = 'bg-accentRed h-full rounded-full transition-all duration-500';

        runAlert.className = 'text-accentRed font-bold';

        runAlert.innerText = 'Interference Alert: Muscle Growth Blunted';

      } else if (state.runHours >= 2.0) {

        runBar.style.width = runPercent + '%';

        runBar.className = 'bg-accentAmber h-full rounded-full transition-all duration-500';

        runAlert.className = 'text-accentAmber font-medium';

        runAlert.innerText = 'Approaching Upper Recovery Limit';

      } else {

        runBar.style.width = runPercent + '%';

        runBar.className = 'bg-accentGreen h-full rounded-full transition-all duration-500';

        runAlert.className = 'text-slate-400 font-medium';

        runAlert.innerText = 'Within Safe Limits';

      }


      // 4. Zone 1-2 Base

      document.getElementById('z12Val').innerText = state.z12Minutes;

      document.getElementById('z12Bar').style.width = Math.min(100, (state.z12Minutes / 120) * 100) + '%';


      // 5. Zone 4-5 High Intensity

      document.getElementById('z45Val').innerText = state.z45Minutes;

      const z45Bar = document.getElementById('z45Bar');

      z45Bar.style.width = Math.min(100, (state.z45Minutes / 20) * 100) + '%';

      if (state.z45Minutes > 20) {

        z45Bar.className = 'bg-accentAmber h-full rounded-full transition-all duration-500';

      } else {

        z45Bar.className = 'bg-accentGreen h-full rounded-full transition-all duration-500';

      }


      // 6. Zone 3 Error Allowance Bar

      const z3Bar = document.getElementById('z3Bar');

      const z3Val = document.getElementById('z3Val');

      const z3Badge = document.getElementById('z3Badge');


      if (state.z3Minutes === 0) {

        z3Bar.style.width = '0%';

        z3Val.innerText = '0 min (Target)';

        z3Val.className = 'font-bold text-accentGreen';

        z3Badge.innerText = 'Zero Gray Zone Stress';

        z3Badge.className = 'text-accentGreen font-semibold';

      } else if (state.z3Minutes < 15) {

        z3Bar.style.width = (state.z3Minutes / 15) * 100 + '%';

        z3Bar.className = 'bg-accentAmber h-full rounded-full transition-all duration-500';

        z3Val.innerText = `${state.z3Minutes} min (Drift)`;

        z3Val.className = 'font-bold text-accentAmber';

        z3Badge.innerText = 'Gray Zone Drift Warning';

        z3Badge.className = 'text-accentAmber font-semibold';

      } else {

        z3Bar.style.width = '100%';

        z3Bar.className = 'bg-accentRed h-full rounded-full transition-all duration-500';

        z3Val.innerText = `${state.z3Minutes} min (Exceeded)`;

        z3Val.className = 'font-bold text-accentRed';

        z3Badge.innerText = 'Autonomic Recovery Penalty';

        z3Badge.className = 'text-accentRed font-bold';

      }


      // 7. Autonomic Readiness & Biometrics

      document.getElementById('hrvVal').innerText = state.hrvLn.toFixed(2);

      document.getElementById('deepSleepVal').innerText = state.deepSleepStr;

      document.getElementById('decouplingVal').innerText = state.decoupling + '%';


      const readinessBadge = document.getElementById('readinessBadge');

      if (state.hrvLn < 3.65 || state.decoupling > 5.0) {

        readinessBadge.innerText = 'AUTONOMIC RECOVERY NEEDED';

        readinessBadge.className = 'text-[11px] font-bold px-2 py-0.5 rounded bg-accentRed/20 text-accentRed border border-accentRed/40';

      } else {

        readinessBadge.innerText = 'CORRIDOR NORMAL';

        readinessBadge.className = 'text-[11px] font-bold px-2 py-0.5 rounded bg-accentGreen/20 text-accentGreen border border-accentGreen/40';

      }

    }


    function toggleMockState() {

      isSimulatedExcess = !isSimulatedExcess;

      const btn = document.getElementById('mockBtn');

      if (isSimulatedExcess) {

        btn.innerText = 'Reset to Nominal';

        btn.className = 'text-[11px] font-semibold px-2.5 py-1 bg-accentRed/20 border border-accentRed/40 rounded-md text-accentRed transition';

        renderState(excessState);

      } else {

        btn.innerText = 'Simulate Excess';

        btn.className = 'text-[11px] font-semibold px-2.5 py-1 bg-subtleDark border border-slate-700 rounded-md text-accentBlue hover:bg-slate-700 transition';

        renderState(baseState);

      }

    }


    // Modal controls for Token

    function openSettings() {

      document.getElementById('settingsModal').classList.remove('hidden');

      document.getElementById('apiTokenInput').value = localStorage.getItem('garmin_personal_token') || '';

    }


    function closeSettings() {

      document.getElementById('settingsModal').classList.add('hidden');

    }


    function saveToken() {

      const token = document.getElementById('apiTokenInput').value.trim();

      if (token) {

        localStorage.setItem('garmin_personal_token', token);

      }

      closeSettings();

    }


    // Initial render

    window.addEventListener('DOMContentLoaded', () => {

      renderState(baseState);

    });
